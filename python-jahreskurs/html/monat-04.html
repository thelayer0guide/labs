Dunder-Methoden
  • __str__ vs. __repr__ — und wann was
  • __len__, __contains__, __getitem__
  • Vergleichsoperatoren: __eq__, __lt__, __le__
  • Arithmetik: __add__, __mul__, __iadd__

class Vektor: def __init__(self, x, y): self.x, self.y = x, y def __repr__(self): return f"Vektor({self.x}, {self.y})" def __add__(self, other): return Vektor(self.x+other.x, self.y+other.y) def __len__(self): return int((self.x**2+self.y**2)**0.5) print(Vektor(3,4) + Vektor(1,1)) # Vektor(4, 5)
