Hintergrund

Bei n Bällen gibt es n*(n-1)/2 mögliche Paare. Bei 100 Bällen: 4950 Checks. Bei 500: 124750. Das wächst quadratisch — O(n²). Spatial Hashing (Uniform Grid): Die Welt wird in Zellen der Größe CELL_SIZE geteilt. Jeder Ball landet in der Zelle seines Mittelpunkts. Beim Kollisionscheck schaut man nur in die eigene und 8 Nachbarzellen — konstante Anzahl Checks pro Ball: O(n).

import pygame, random, time
pygame.init()
W, H = 900, 650
screen = pygame.display.set_mode((W, H))
clock  = pygame.time.Clock()
font   = pygame.font.SysFont("Arial", 14)

CELL = 50  # Grid-Zellgröße
R    = 12  # Radius aller Bälle

class Ball:
    def __init__(self):
        self.pos = pygame.math.Vector2(random.randint(R,W-R), random.randint(R,H-R))
        self.vel = pygame.math.Vector2(random.uniform(-150,150), random.uniform(-150,150))

    def update(self, dt):
        self.pos += self.vel * dt
        if self.pos.x - R < 0:  self.pos.x = R;   self.vel.x =  abs(self.vel.x)
        if self.pos.x + R > W:  self.pos.x = W-R; self.vel.x = -abs(self.vel.x)
        if self.pos.y - R < 0:  self.pos.y = R;   self.vel.y =  abs(self.vel.y)
        if self.pos.y + R > H:  self.pos.y = H-R; self.vel.y = -abs(self.vel.y)

def build_grid(balls):
    grid = {}
    for b in balls:
        key = (int(b.pos.x // CELL), int(b.pos.y // CELL))
        grid.setdefault(key, []).append(b)
    return grid

def get_candidates(grid, b):
    cx, cy = int(b.pos.x // CELL), int(b.pos.y // CELL)
    for dx in (-1, 0, 1):
        for dy in (-1, 0, 1):
            yield from grid.get((cx+dx, cy+dy), [])

def resolve(a, b):
    d = b.pos - a.pos
    dist = d.length()
    if dist < 2*R and dist > 0.01:
        n = d / dist
        overlap = 2*R - dist
        a.pos -= n * overlap/2; b.pos += n * overlap/2
        vrel = (b.vel - a.vel).dot(n)
        if vrel < 0:
            j = -(1.8) * vrel / 2
            a.vel -= n*j; b.vel += n*j

balls = [Ball() for _ in range(80)]
use_grid  = True
checks    = 0

running = True
while running:
    dt = min(clock.tick(60)/1000, 0.033)
    for event in pygame.event.get():
        if event.type == pygame.QUIT: running = False
        if event.type == pygame.KEYDOWN and event.key == pygame.K_SPACE: use_grid = not use_grid
        if event.type == pygame.KEYDOWN and event.key == pygame.K_UP:    balls += [Ball() for _ in range(20)]

    for b in balls: b.update(dt)

    checks = 0
    if use_grid:
        grid = build_grid(balls)
