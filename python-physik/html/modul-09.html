Federn, Pendel & Hooke

Cheat Sheets
python
Lernziele
  • Hooke's Law als Federkraft implementieren: F = -k*(|x|-L₀)*n
  • Dämpfung der Feder durch Relativgeschwindigkeit entlang der Federachse
  • Ein einfaches und ein Doppelpendel simulieren
  • Instabilität bei zu kleinem dt und hartem k erkennen und beheben
Hintergrund

Hooke's Law: Eine Feder verbindet zwei Punkte A und B. Ruhelänge L₀. Aktuelle Länge L = |B-A|. Ausdehnung x = L - L₀. Federkraft auf A: F_A = k * x * normalize(B-A). Kraft auf B: F_B = -F_A. Dämpfung: Relative Geschwindigkeit der Punkte entlang der Federachse: v_rel = (vB-vA)·n; F_damp = -d * v_rel * n. Kombination: F_gesamt = (k*x + d*v_rel) * n.

Numerische Stabilität: Bei großem k oder kleinem dt beginnt die Simulation zu schwingen und explodiert. Lösung: k nicht zu groß wählen (< 2000 für dt=1/60), oder Verlet-Integration nutzen (Modul 11). Doppelpendel: zwei Pendel hintereinandergehängt — chaotisches System, extrem sensitiv gegenüber Anfangsbedingungen. Kleine Änderungen in θ₁, θ₂ führen zu komplett anderen Trajektorien (Chaos-Theorie).

import pygame, math
pygame.init()
W, H = 800, 600
screen = pygame.display.set_mode((W, H))
clock  = pygame.time.Clock()
font   = pygame.font.SysFont("Arial", 14)

# ── Feder-System ─────────────────────────────────────────────────────
class Spring:
    def __init__(self, p1, p2, k=300, damping=8, rest=None):
        self.p1   = pygame.math.Vector2(p1)
        self.p2   = pygame.math.Vector2(p2)
        self.v1   = pygame.math.Vector2(0, 0)
        self.v2   = pygame.math.Vector2(0, 0)
        self.k    = k
        self.damp = damping
        self.rest = rest or (pygame.math.Vector2(p2)-pygame.math.Vector2(p1)).length()
        self.anchored1 = True   # p1 fixiert

    def update(self, dt):
        delta = self.p2 - self.p1
        dist  = delta.length()
        if dist < 0.001: return
        n      = delta / dist
        x      = dist - self.rest                      # Ausdehnung
        v_rel  = (self.v2 - self.v1).dot(n)            # Relativ-Geschw. entlang Feder
        force  = (self.k * x + self.damp * v_rel)      # Hooke + Dämpfung
        if not self.anchored1:
            self.v1  +=  n * force * dt
        self.v2      += -n * force * dt
        self.v2.y    +=  600 * dt   # Schwerkraft auf p2
        self.p2      += self.v2 * dt
        # Wände
        if self.p2.x < 10:  self.p2.x = 10;  self.v2.x =  abs(self.v2.x)*0.7
        if self.p2.x > W-10:self.p2.x = W-10;self.v2.x = -abs(self.v2.x)*0.7
        if self.p2.y > H-10:self.p2.y = H-10;self.v2.y = -abs(self.v2.y)*0.7

    def draw(self, surf):
        p1i = (int(self.p1.x), int(self.p1.y))
        p2i = (int(self.p2.x), int(self.p2.y))
        pygame.draw.line(surf, (63,185,80), p1i, p2i, 2)
        pygame.draw.circle(surf, (63,185,80),  p1i, 8)
        pygame.draw.circle(surf, (88,166,255), p2i, 12)

# ── Doppelpendel ─────────────────────────────────────────────────────
class DoublePendulum:
    def __init__(self, ox, oy, L1=130, L2=130, th1=math.pi*0.9, th2=math.pi*0.6):
        self.ox, self.oy = ox, oy
        self.L1, self.L2 = L1, L2
        self.th1, self.th2 = th1, th2
        self.w1 = self.w2 = 0.0
        self.trail = []

    def update(self, dt):
        g = 9.81 * 60  # skaliert
        m = 1.0
        th1, th2, w1, w2 = self.th1, self.th2, self.w1, self.w2
        d = th1 - th2
        denom1 = (2*m + m - m*math.cos(2*d)) * self.L1
        denom2 = (2*m + m - m*math.cos(2*d)) * self.L2
        if abs(denom1) < 1e-9 or abs(denom2) < 1e-9: return
        a1 = (-g*(2*m+m)*math.sin(th1) - m*g*math.sin(th1-2*th2) - 2*math.sin(d)*m*(w2**2*self.L2 + w1**2*self.L1*math.cos(d))) / denom1
        a2 = (2*math.sin(d)*(w1**2*self.L1*(2*m) + g*(2*m+m)*math.cos(th1) + w2**2*self.L2*m*math.cos(d))) / denom2
        self.w1 += a1 * dt; self.w2 += a2 * dt
        self.th1 += self.w1 * dt; self.th2 += self.w2 * dt
        x2 = self.ox + self.L1*math.sin(th1) + self.L2*math.sin(th2)
        y2 = self.oy + self.L1*math.cos(th1) + self.L2*math.cos(th2)
        self.trail.append((int(x2),int(y2)))
        if len(self.trail) > 500: self.trail.pop(0)

    def draw(self, surf):
        x1 = self.ox + self.L1*math.sin(self.th1)
        y1 = self.oy + self.L1*math.cos(self.th1)
        x2 = x1 + self.L2*math.sin(self.th2)
        y2 = y1 + self.L2*math.cos(self.th2)
        if len(self.trail) > 1:
            pygame.draw.lines(surf, (30,60,50), False, self.trail, 1)
        pygame.draw.line(surf, (125,133,144),(self.ox,self.oy),(int(x1),int(y1)),2)
        pygame.draw.line(surf, (125,133,144),(int(x1),int(y1)),(int(x2),int(y2)),2)
        pygame.draw.circle(surf,(255,123,114),(int(x1),int(y1)),10)
        pygame.draw.circle(surf,(255,166,87), (int(x2),int(y2)),10)

spring = Spring((400,50),(400,200))
pendel = DoublePendulum(700, 150)
mode   = "spring"

running = True
while running:
    dt = min(clock.tick(60)/1000, 0.02)
    for event in pygame.event.get():
        if event.type == pygame.QUIT: running = False
        if event.type == pygame.KEYDOWN and event.key == pygame.K_SPACE:
            mode = "pendel" if mode=="spring" else "spring"
        if event.type == pygame.MOUSEBUTTONDOWN and mode=="spring":
            spring.p2 = pygame.math.Vector2(event.pos); spring.v2.xy = 0,0

    if mode == "spring": spring.update(dt)
    else: pendel.update(dt)

    screen.fill((13,17,23))
    if mode == "spring":
        spring.draw(screen)
        screen.blit(font.render("SPACE=Doppelpendel  Klick=Ball ziehen", True,(125,133,144)),(10,10))
    else:
        pendel.draw(screen)
        screen.blit(font.render("SPACE=Feder-System  (Doppelpendel — chaotisch)", True,(125,133,144)),(10,10))
    pygame.display.flip()

pygame.quit()

Das Doppelpendel ist ein klassisches Beispiel für deterministisches Chaos — minimal andere Startwerte → komplett andere Trajektorie.

Aufgabe

Implementiere ein Feder-Netz aus 6+ Punkten die durch Federn verbunden sind (Gitter oder Polygon). Ziehe an einem Punkt und beobachte die Schwingung.

Anforderungen
  1. Federkraft: k*(dist-rest)*n korrekt
  2. Dämpfung: d*v_rel*n korrekt entlang Federachse
  3. Fixierte und freie Punkte unterschieden
  4. Numerische Stabilität: dt maximal geclampt
  5. Doppelpendel-Bewegungsgleichungen korrekt implementiert
Akzeptanzkriterien
  • Feder schwingt und kommt durch Dämpfung zur Ruhe
  • Doppelpendel zeigt chaotisches Verhalten
  • System explodiert nicht bei normalen k/dt-Werten
